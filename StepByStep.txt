# ---------- 3. Models (already in .\Models, no re-download needed) ----------
# pocket-tts-farsi-v2 (phoneme model + tokenizer + local yaml):
#   Models\pocket-tts-farsi-v2\model.safetensors
#   Models\pocket-tts-farsi-v2\tokenizer_ph.model

Get-ChildItem .\Models -Force
Get-ChildItem "E:\AI\pocket-tts\Models\pocket-tts-farsi-v2" -Force

# ---------- 4. Synthesize (phoneme input! Persian script -> silence) ----------
# model_farsi_local.yaml at repo root = copy of Models\pocket-tts-farsi-v2\model.yaml
# with weights_path/tokenizer_path rewritten to local E:/AI/... absolute paths
# so no HuggingFace download happens at runtime.
# "salAm hAle SomA Cetor ?ast" = phonemes for "سلام، حال شما چطور است؟" (see Models\pocket-tts-farsi-v2\README.md phonemise()).
# --voice-sec 5 trims the prompt (training cap was 5s; longer prompts replay the prompt).
# --no-normalize-text = correct here ONLY because input is already phonemes (else keep default).
.\.venv\Scripts\python.exe -m training.farsi.synthesize `
    --config .\model_farsi_local.yaml `
    --voice .\training\farsi\examples\01_greeting.wav `
    --voice-sec 5 `
    --no-normalize-text `
    --text "salAm hAle SomA Cetor ?ast" `
    --out .\test_fa.wav


.\.venv\Scripts\python.exe -m training.farsi.synthesize --config .\model_farsi_local.yaml --voice .\Models\pocket-tts-farsi-v2\samples\prompt_hello.wav --voice-sec 5 --no-normalize-text --text "salAm hAle SomA Cetor ?ast" --out .\test_fa.wav

# ---------- 5. Serve (optional) ----------
.\.venv\Scripts\python.exe -m pocket_tts.main serve
